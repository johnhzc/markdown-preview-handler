4. 代码块

def schlieren_focus(f, n, r):
    """液滴透镜焦距估算（近轴近似）"""
    R = 2 * r                      # 曲率半径
    f_lens = R / (2 * (n - 1))     # 薄透镜公式
    return f_lens + f
